***Match_PlayLoop***
***
//Replay_SaveInterface();
Dynamics::Compute();

// One tick of the launch clock. Counted rather than measured in ms because what a firing action
// instance needs between a disarm and the next arm is a read of its own, and it reads once per
// pass — see C_LaunchSlotRearmPasses and LaunchSlotReady.
G_LaunchLoopPass += 1;

// Close the PvP window opened for a trampoline launch (opened on touch, collapsed on the fire echo)
if (G_TrampolinePvPWindowEnd > 0 && Now > G_TrampolinePvPWindowEnd) {
    G_TrampolinePvPWindowEnd = 0;
    G_TrampolinePvPInFlight = 0;	// window closed -> nothing can still be waiting on it (self-heals a lost-echo leak)
    UsePvPWeapons = G_TrampolinePvPRestore;
    G_LastUsePvPWeapons = UsePvPWeapons;
}

// If a new event edition was loaded, notify the UI
if (ObsEvents::Loop()) {
    EventsUI::UpdateActiveEvents();
}

EventsUI::Loop();

if (LastUpdate + C_SettingsUpdateInterval < Now) {
    LastUpdate = Now;
    UpdateSettings();
}

if (Settings::GetForceInvisibleSkin()) {
    Obstacle::UpdatePlayersSkin(G_InvisibleSkinId);
} else {
    Obstacle::UpdatePlayersSkin(NullId);
}

if (Settings::GetEnableRecords()) {
    Layers::Attach("Records");
} else {
    Layers::Detach("Records");
}

if (EndTime > 0 && Now > EndTime) {
    if (EventsUI::RequestedRestart()) {
        EventsUI::BeginRestartMap();
        CreateUI(False);
        EventsUI::EndRestartMap();
    } else {
        Message::SendBigMessage(_("Time limit reached"),    2000, 3, CUIConfig::EUISound::Silence, 0);
        MB_Private_StopMatch();
    }
}

if ((EndTime - StartTime) > 60000 && Now > EndTime - 300500 && Now < EndTime - 290000 && Now > NextLowTimeMessage) {
    UIManager.UIAll.SendChat("Warning: the map ends in $<$f805$> minutes!");
    NextLowTimeMessage = Now + 305000;
}
declare CSmModeEvent[] PendingRespawns;
declare CSmModeEvent[] PendingCaptures;

foreach (Event in PendingEvents) {
    switch (Event.Type) {
				case CSmModeEvent::EType::OnShoot: {
					//Dynamics::Info();

				}
        case CSmModeEvent::EType::OnArmorEmpty: {
            // Discard the defaut behaviour (e.g Player is eliminated)
            Events::Invalid(Event);

            declare CSmPlayer Victim = Null;

            switchtype (Event.VictimEntity) {
                case CSmPlayer: {
                    Victim = (Event.VictimEntity as CSmPlayer);
                }
                default: {
                    continue;
                }
            }

            // Dont respawn if the player has the offzone item
            if (Victim.IsInOffZone) {
                declare ItemBuffs = Player::getBuffTimes(Victim);
                if (ItemBuffs.existskey(Constants::C_ItemOffzone) && ItemBuffs[Constants::C_ItemOffzone] >= Now) {
                    continue;
                } 
            }

            Obstacle::ObsRespawnPlayer(Victim);
        }

        case CSmModeEvent::EType::OnHit: {
            // Disable the loose of armor when hit
						 Dynamics::OnHit(Event.Victim, Event.VictimEntity);
            Events::Invalid(Event);
        }

        case CSmModeEvent::EType::OnActionCustomEvent : {
            // :'(

            // Events from the trampoline action: a press from a bound key slot, or the fire echo
            // — the variant the action launched with.
            if (Event.Param1 == "trampoline") {
                declare Payload = "?";
                if (Event.Param2.count > 0) Payload = Event.Param2[0];
                if (Payload == "press") {
                    // A bound key slot reporting its own activation. The event's own slot fields
                    // are useless here (ShooterActionSlot is always Slot_A, ShooterActionId is
                    // empty) and the OnActionEvent that carries a key is only delivered for inputs
                    // with no action bound — so before the per-key models existed, nothing could
                    // tell two granted keys apart and both fell through to the single-granted-key
                    // rescue, which is why granting a second key killed both.
                    //
                    // Event.ActionId names the model that sent the echo, and each key slot runs
                    // its own copy, so the key is now in the event itself.
                    switchtype (Event.ShooterEntity) {
                        case CSmPlayer: {
                            declare Presser = (Event.ShooterEntity as CSmPlayer);
                            declare ModelKey = LaunchKeyFromActionId(Event.ActionId);
                            // An echo from any other model is ignored: only the key slots are ever
                            // bound, so only they can send a press. Guessing a key here (the old
                            // "the section grants exactly one key" rescue) is what broke as soon
                            // as a section granted two.
                            if (ModelKey > 0) {
                                TryLaunchOnKey(Presser, ModelKey);
                            }
                        }
                        default: { DoNothing(); }
                    }
                }
                else {
                    switchtype (Event.ShooterEntity) {
                        case CSmPlayer: {
                            declare Shooter = (Event.ShooterEntity as CSmPlayer);
                            // Low 16 bits only: an action instance carrying an energy gauge
                            // reads its Variant back as EnergyMax*65536 + what was armed, and
                            // an older Gbx echoes that raw value. Masking here keeps the mode
                            // working with either build.
                            declare EchoedVariant = TL::ToInteger(Payload) % 65536;
                            // Any encoding base: the world-compass one, the aim-relative one or the
                            // movement-relative one. Pads and keys both pick freely between them,
                            // and every one of them must disarm its slot on the echo.
                            if (EchoedVariant >= Constants::C_TrampolineVariantBase
                                && EchoedVariant <= Constants::C_TrampolineVariantBaseMovement + Constants::C_TrampolineVariantPackedMax) {
                                // Normal launch: hand the slot back so the action can fire again
                                // next touch or key press (pad slot and key slots share this
                                // bookkeeping). Which slot fired is in the event: every key slot
                                // runs its own copy of the action and the pad slot runs the shared
                                // one, so the model that echoed names the slot it was loaded into
                                // — 0, the pad slot, being both the shared model's answer and the
                                // right answer for an echo we cannot place.
                                //
                                // Resolved off the arm stamps instead, this released whichever
                                // launch was armed first, which is the wrong one as soon as a pad
                                // touch and a key press are outstanding together (possible
                                // whenever C_LaunchFireOnPadSlot is False): the slot that really
                                // fired stayed marked armed until its timeout, while a slot still
                                // holding a launch was disarmed under it.
                                ReleaseLaunchSlot(Shooter, LaunchKeyFromActionId(Event.ActionId));
                                RetireLaunchInFlight();
                            } else {
                                // An echo that is not a launch encoding: the action fired on a
                                // value the mode never wrote. A Gbx built before the Variant
                                // masking does exactly this — a disarmed instance reads its own
                                // EnergyMax*65536, which is not 0, so it "fires" once on it.
                                //
                                // Ignore it. Reloading the slot here produced an endless
                                // reload -> fresh instance -> stray echo -> reload storm in-game
                                // (a permanent "fired 419430400" flood that also wiped every
                                // armed launch before it could fire), while the stray echo on its
                                // own is harmless: the instance goes quiet after that one shot.
                                //
                                // Nothing is retired from the in-flight count either: no launch of
                                // ours resolved here, and the launches that are genuinely still
                                // outstanding need the PvP window to stay open until they fire.
                                DoNothing();
                            }
                        }
                        default: { DoNothing(); }
                    }
                }
            }
        }

        case CSmModeEvent::EType::OnPlayerTouchesObject: {
						declare Boolean PerformEliminate = Dynamics::CheckEliminatingObject(Event.Object, "touch", Event.Player);
							
						if(PerformEliminate) {
							if(Player::getLastCheckpoint(Event.Player) == NullId && (ServerModeName == "" || ServerLogin == "")) {
								Dynamics::ForceResetAll();
							}
							Obstacle::ObsRespawnPlayer(Event.Player);
						}
						
						if(Dynamics::IsDynamic(Event.Object)) {
								Events::Invalid(Event);
								continue;
						}

						// Trampoline item: arm the trampoline action to launch the toucher
						if (Event.Object != Null && Event.Player != Null) {
							declare Text ItemName for Event.Object = "";
							if (TL::Find("Trampoline", ItemName, False, True)) {
								// Arming guard: the OnPlayerTouchesObject event fires across the whole TriggerShape,
									// but arming must reach every point in it (a 6x6 pad corner is ~4.24m from the anchor).
									// Kept below a respawn distance so it still drops stale touches queued across a respawn.
								if (ML::Distance(Event.Object.AnchorLandmark.Position, Event.Player.Position) <= Constants::C_TrampolineTouchMaxDist) {
									// Launch gates, set per anchor by the mapper (copied onto the object, clamped, in Items::Place)
									declare Boolean TrampolineWaitGround for Event.Object = Constants::C_TrampolineDefaultWaitGround;
									declare Integer TrampolineDelay for Event.Object = Constants::C_TrampolineDefaultDelay;

									// Delay timer: touch events repeat while the player overlaps the pad, so a gap
									// in events means they left the pad and the timer restarts on the next touch.
									// On wait-ground pads the delay counts from landing instead: airborne touches
									// park the timer (-1) and it starts on the first grounded touch.
									declare Integer[Ident] TrampolineTouchStart for Event.Player = [];
									declare Integer[Ident] TrampolineTouchLast for Event.Player = [];
									declare TimerRunning = !TrampolineWaitGround || Event.Player.IsTouchingGround;
									if (!TimerRunning) {
										TrampolineTouchStart[Event.Object.Id] = -1;
									} else if (!TrampolineTouchStart.existskey(Event.Object.Id) || TrampolineTouchStart[Event.Object.Id] < 0 || Now > TrampolineTouchLast[Event.Object.Id] + Constants::C_TrampolineTouchResetGap) {
										TrampolineTouchStart[Event.Object.Id] = Now;
									}
									TrampolineTouchLast[Event.Object.Id] = Now;

									// Gate closed while airborne on a wait-ground pad, and until the delay has
									// passed since the timer started (pad contact / landing respectively)
									declare LaunchGatesOpen = TimerRunning;
									if (TimerRunning && Now < TrampolineTouchStart[Event.Object.Id] + TrampolineDelay) {
										LaunchGatesOpen = False;
									}

									// Pad-wide cooldown: the pad that fires stamps the player, and no trampoline
									// launches them again until it passes -- which is also what keeps a pad from
									// refiring every tick while they stand on it. Only an actual launch stamps it,
									// so a pad left at strength 0 never silences the ones around it.
									// LaunchSlotReady is in the same condition as the cooldown so a touch the
									// pad slot cannot take yet stamps nothing: touches keep flowing every tick,
									// so the launch goes out on the first one the slot is free for.
									declare Integer TrampolineCooldownEnd for Event.Player = 0;
									if (LaunchGatesOpen && Now > TrampolineCooldownEnd && LaunchSlotReady(Event.Player, 0)) {
										// Launch parameters from the anchor metadata (copied onto the
										// object, clamped, in Items::Place)
										declare Integer TrampolineUp for Event.Object = Constants::C_TrampolineDefaultUp;
										declare Integer TrampolineDown for Event.Object = Constants::C_TrampolineDefaultDown;
										declare Integer TrampolineCompass for Event.Object = Constants::C_TrampolineDefaultCompass;
										declare Integer TrampolineFrame for Event.Object = Constants::C_TrampolineDefaultFrame;
										declare Integer TrampolineHoriz for Event.Object = Constants::C_TrampolineDefaultHoriz;

										// Policy: optionally suppress the down blast when an up blast fires too
										declare LaunchDown = Tools::LaunchDownStrength(TrampolineUp, TrampolineDown);

										if (TrampolineUp > 0 || LaunchDown > 0 || TrampolineHoriz > 0) {
											// Encoding and direction calibration both live in LaunchVariant, in the
											// frame this pad picked (world compass, the player's aim, or their travel)
											declare TrampolineVariant = LaunchVariant(TrampolineFrame,
												TrampolineUp, TrampolineDown, TrampolineCompass, TrampolineHoriz);
											declare Integer TrampolineCooldown for Event.Object = Constants::C_TrampolineDefaultCooldown;
											TrampolineCooldownEnd = Now + TrampolineCooldown;
											// Slot index 0 = the pad slot; opens the PvP window and arms the action
											ArmLaunch(Event.Player, 0, TrampolineVariant);
										}
									}
								}
								// Discard (not Valid/PassOn): the trampoline is a stand-on pad, not a pickup.
								// PassOn makes the engine run its default touch effect (flash, sound, hide the
								// object) every tick of overlap; Discard suppresses it while touches keep flowing.
								Events::Invalid(Event);
								continue;
							}
						}

           if (Event.Object != Null && Event.Player != Null && Items::PickupItems(Event)) {
								
                Events::Valid(Event);
            }
            else {
                Events::Invalid(Event);
            }
        }

        case CSmModeEvent::EType::OnCommand: {
            if (Event.CommandName == "Command_BeginSurvival") {
                if (Settings::GetScoreMode() != Constants::C_Score_Survival) {
                    ObsScores::Survival_Begin(Event.CommandValueInteger);
                    MB_Private_StopMap();
                }
            }
            else if (Event.CommandName == "Command_EndSurvival") {
                if (Settings::GetScoreMode() == Constants::C_Score_Survival) {
                    ObsScores::Survival_ForceEnd();
                }
            }
        }

        case CSmModeEvent::EType::OnPlayerRequestRespawn: {
            PendingRespawns.add(Event);
        }

        case CSmModeEvent::EType::OnPlayerTriggersSector: {
            PendingCaptures.add(Event);
        }

        // No OnActionEvent case: the engine delivers it only for inputs that NO action is bound
        // to (verified in-game — pressing a granted, bound key produces the action's own press
        // echo and no OnActionEvent, pressing an ungranted key produces the reverse). So it can
        // never report a launch key that is actually usable, and using it to launch would fire on
        // keys the section hasn't granted. The press echo's Event.ActionId is the identity source.
        default: {
            Events::Valid(Event);
        }

    }
}

// ---------------------------------- //
// Functions
// ---------------------------------- //
// ---------------------------------- //
Void DoNothing(){}

// Load without ActionBind — the action must tick unbound and fire only from the variant the mode
// sets (Slot_D is free: ObstacleLib loads no actions by default). Reloaded on every spawn edge and
// on garbage echoes to replace the engine-corrupted instance.
Void LoadTrampolineAction(CSmPlayer _Player) {
    if (G_TrampolineActionId == NullId) return;
    ActionLoad(_Player, CSmMode::EActionSlot::Slot_D, G_TrampolineActionId);
    // Clear the slot variant so the fresh instance can't fire on a stale or
    // garbage value (armed-at-death leftovers, post-respawn corruption)
    ActionSetVariant(_Player, CSmMode::EActionSlot::Slot_D, 0);
}

// ---------------------------------- //
// Launch abilities: the trampoline launch, triggered by the player on keys 1..4 instead of
// by a pad. A checkpoint (or the start spawn) grants and configures them for its section.
//
// Keys 1..4 (Activable1..4) live in Slot_E..Slot_H and run the same action model the pads
// use in Slot_D. A slot is loaded inert (variant 0) and only *bound* while the player's
// current section grants that key, so the HUD shows exactly the abilities they have.
// Pressing a bound key makes the mode write a launch variant into that slot — the same
// arming a pad touch does, PvP window included — and the fire echo disarms it again.
// Slot index used below: 0 = the pad slot, 1..4 = key 1..4.
// ---------------------------------- //
CSmMode::EActionSlot LaunchSlot(Integer _SlotIndex) {
    switch (_SlotIndex) {
        case 1: return CSmMode::EActionSlot::Slot_E;
        case 2: return CSmMode::EActionSlot::Slot_F;
        case 3: return CSmMode::EActionSlot::Slot_G;
        case 4: return CSmMode::EActionSlot::Slot_H;
    }
    return CSmMode::EActionSlot::Slot_D;
}

CSmMode::EActionInput LaunchInput(Integer _Key) {
    switch (_Key) {
        case 2: return CSmMode::EActionInput::Activable2;
        case 3: return CSmMode::EActionInput::Activable3;
        case 4: return CSmMode::EActionInput::Activable4;
    }
    return CSmMode::EActionInput::Activable1;
}

// The key that a press echo came from, read off the *model* that sent it. Each key slot is
// loaded with its own copy of the trampoline action (C_LaunchActionModels), and every action
// event names its model in Event.ActionId — the one piece of identity a bound action's echo does
// carry (verified in-game: ActionId "Trampoline.Action.Gbx" while ShooterActionId was empty and
// ShooterActionSlot was Slot_A). 0 = not one of the key models (e.g. the pad slot's fire echo).
Integer LaunchKeyFromActionId(Text _ActionId) {
    if (_ActionId == "") { return 0; }
    declare Key = 0;
    foreach (ModelName in Constants::C_LaunchActionModels) {
        Key += 1;
        if (ModelName == _ActionId) {
            return Key;
        }
    }
    return 0;
}

// Editor compass (N..NW clockwise) -> action compass index (45 deg steps, +Z toward +X)
Integer LaunchActionCompass(Integer _EditorCompass) {
    declare Steps = _EditorCompass;
    if (Constants::C_TrampolineCompassMirror) {
        Steps = (8 - _EditorCompass) % 8;
    }
    return (Steps + Constants::C_TrampolineCompassOffset) % 8;
}

// Editor launch direction (Forward..Forward-Left clockwise) -> the 45 deg steps the action
// turns the owner's aim by, in the action's own rotation sense. C_TrampolineCompassOffset is
// deliberately not applied here: it calibrates where the editor's "N" sits in the world, while
// step 0 must simply mean "straight ahead" wherever the player is looking.
Integer LaunchRelativeSteps(Integer _EditorDir) {
    if (Constants::C_LaunchRelativeMirror) {
        return (8 - _EditorDir) % 8;
    }
    return _EditorDir % 8;
}

// The variant base that makes the action read the dir digit in the frame the mapper picked.
// Anything unrecognised falls back to the world frame, which is what a pad without the
// metadata has always meant.
Integer LaunchFrameBase(Integer _Frame) {
    if (_Frame == Constants::C_LaunchFrameOrientation) {
        return Constants::C_TrampolineVariantBaseRelative;
    }
    if (_Frame == Constants::C_LaunchFrameMovement) {
        return Constants::C_TrampolineVariantBaseMovement;
    }
    return Constants::C_TrampolineVariantBase;
}

// Editor direction index -> the 45 deg steps the action applies, in that frame's rotation
// sense. The world frame turns around +Z and carries the compass calibration; the two
// player frames turn around the player with one shared sense (see C_LaunchRelativeMirror).
Integer LaunchFrameSteps(Integer _Frame, Integer _EditorDir) {
    if (_Frame == Constants::C_LaunchFrameOrientation || _Frame == Constants::C_LaunchFrameMovement) {
        return LaunchRelativeSteps(_EditorDir);
    }
    return LaunchActionCompass(_EditorDir);
}

// The whole of encoding v4 (must match the action): the base names the frame the dir digit is
// measured in — a world compass direction, a 45 deg step from where the player looks, or one from
// where they are travelling — over base + up*1000 + down*100 + dir*10 + horiz.
Integer LaunchVariant(Integer _Frame, Integer _Up, Integer _Down, Integer _Dir, Integer _Horiz) {
    return LaunchFrameBase(_Frame)
        + _Up * 1000 + Tools::LaunchDownStrength(_Up, _Down) * 100 + LaunchFrameSteps(_Frame, _Dir) * 10 + _Horiz;
}

// How many charges the player enters a section with, in the order the weapon rules apply: carry
// what was left over, start full on a key this section is the first to grant (and on a respawn,
// where weapon ammo is refilled too), then let the mapper's Refill / Empty override. Called by
// LoadLaunchConfig once the section's own settings are in place, with what the previous section
// had.
Void InitLaunchCharges(CSmPlayer _Player, Boolean[] _PreviousOn, Integer[] _PreviousCharges, Boolean _Respawned) {
    declare Integer[] LaunchAmount for _Player = [];
    declare Boolean[] LaunchRefill for _Player = [];
    declare Boolean[] LaunchEmpty for _Player = [];
    declare Integer[] LaunchCharges for _Player = [];

    for (Key, 1, Constants::C_LaunchKeyCount) {
        declare Index = Key - 1;
        declare Max = LaunchAmount[Index];

        declare Charges = 0;
        if (_PreviousCharges.count > Index) {
            Charges = _PreviousCharges[Index];
        }
        declare WasGranted = _PreviousOn.count > Index && _PreviousOn[Index];
        if (!WasGranted || (_Respawned && S_FullAmmoSpawn)) {
            Charges = Max;
        }
        if (LaunchRefill[Index]) {
            Charges = Max;
        }
        if (LaunchEmpty[Index]) {
            Charges = 0;
        }

        LaunchCharges[Index] = ML::Clamp(Charges, 0, Max);
    }
}

// Does any section on this map grant a launch key at all? Answered once at map start, because
// both halves of the feature cost something on every map otherwise: the HUD layer instantiates
// ~210 controls per client (160 bar quads alone) and caches them all before its first frame, and
// LoadLaunchConfig re-reads the anchor's launch config on every spawn and every checkpoint.
Boolean MapHasLaunchKeys() {
    foreach (Landmark in MapLandmarks) {
        declare metadata Integer[] Meta_LaunchConfig for Landmark = [];

        // Only the On field of each key block is worth reading here — layout note in Constants.
        for (Key, 1, Constants::C_LaunchKeyCount) {
            declare Index = (Key - 1) * Constants::C_LaunchFieldCount + Constants::C_LaunchFieldOn;
            if (Meta_LaunchConfig.count > Index && Meta_LaunchConfig[Index] == 1) {
                return True;
            }
        }
    }
    return False;
}

// Read the launch settings off the landmark governing the player's current section — the
// last checkpoint they captured, or the spawn before the first one. No landmark (spectator,
// pre-spawn, legacy map without the metadata) leaves every key ungranted.
//
// _Respawned tells the two ways into a section apart: capturing a checkpoint carries the
// charges over (weapon ammo does the same), while respawning tops them back up, the way
// S_FullAmmoSpawn refills weapon ammo on every respawn.
// Every launch array back to "nothing granted", so a section only has to write what it grants.
// Split out of LoadLaunchConfig because the two together exceed the linter's statement cap.
Void ResetLaunchConfig(CSmPlayer _Player) {
    declare Boolean[] LaunchOn for _Player = [];
    declare Integer[] LaunchUp for _Player = [];
    declare Integer[] LaunchDown for _Player = [];
    declare Integer[] LaunchSide for _Player = [];
    declare Integer[] LaunchDir for _Player = [];
    declare Integer[] LaunchFrame for _Player = [];
    declare Integer[] LaunchCooldown for _Player = [];
    declare Boolean[] LaunchGround for _Player = [];
    declare Integer[] LaunchReadyAt for _Player = [];
    declare Integer[] LaunchAmount for _Player = [];
    declare Integer[] LaunchGain for _Player = [];
    declare Boolean[] LaunchRefill for _Player = [];
    declare Boolean[] LaunchEmpty for _Player = [];
    declare Integer[] LaunchCharges for _Player = [];
    declare Integer[] LaunchChargeAt for _Player = [];
    // Whether anything below can draw or fire. Cleared here so LoadLaunchConfig's no-landmark
    // exit leaves it False, which is what stops the per-tick work for spectators and legacy maps.
    declare Boolean LaunchAnyOn for _Player = False;

    LaunchOn = Tools::LaunchFalseSlots();
    LaunchUp = Tools::LaunchZeroSlots();
    LaunchDown = Tools::LaunchZeroSlots();
    LaunchSide = Tools::LaunchZeroSlots();
    LaunchDir = Tools::LaunchZeroSlots();
    LaunchFrame = Tools::LaunchZeroSlots();
    LaunchCooldown = Tools::LaunchZeroSlots();
    LaunchGround = Tools::LaunchFalseSlots();
    LaunchReadyAt = Tools::LaunchZeroSlots();
    LaunchAmount = Tools::LaunchZeroSlots();
    LaunchGain = Tools::LaunchZeroSlots();
    LaunchRefill = Tools::LaunchFalseSlots();
    LaunchEmpty = Tools::LaunchFalseSlots();
    LaunchCharges = Tools::LaunchZeroSlots();
    LaunchChargeAt = Tools::LaunchZeroSlots();
    LaunchAnyOn = False;
}

Void LoadLaunchConfig(CSmPlayer _Player, Boolean _Respawned) {
    declare Boolean[] LaunchOn for _Player = [];
    declare Integer[] LaunchUp for _Player = [];
    declare Integer[] LaunchDown for _Player = [];
    declare Integer[] LaunchSide for _Player = [];
    declare Integer[] LaunchDir for _Player = [];
    declare Integer[] LaunchFrame for _Player = [];
    declare Integer[] LaunchCooldown for _Player = [];
    declare Boolean[] LaunchGround for _Player = [];
    declare Integer[] LaunchAmount for _Player = [];
    declare Integer[] LaunchGain for _Player = [];
    declare Boolean[] LaunchRefill for _Player = [];
    declare Boolean[] LaunchEmpty for _Player = [];
    declare Integer[] LaunchCharges for _Player = [];
    declare Boolean LaunchAnyOn for _Player = False;

    // What the section being left had: the carry-over rule needs both, and the arrays are about
    // to be overwritten.
    declare PreviousOn = LaunchOn;
    declare PreviousCharges = LaunchCharges;

    ResetLaunchConfig(_Player);

    declare Section = Player::GetSectionAnchor(_Player);
    if (!G_MapHasLaunchKeys || !MapLandmarks.existskey(Section)) {
        // Nothing granted: no launch metadata anywhere on the map, or no landmark governing this
        // player (spectator, pre-spawn, legacy map). Publish the zeroes once so a strip left over
        // from the previous section goes away, rather than leaving the last setup on screen. The
        // reset above still runs on the map-wide exit, so leaving a launch map for one without
        // launch keys unbinds what the last one granted.
        LaunchKeys::PublishConfig(_Player);
        LaunchKeys::PublishCharges(_Player);
        return;
    }
    declare Anchor <=> MapLandmarks[Section];
    declare metadata Integer[] Meta_LaunchConfig for Anchor = [];

    // One array per anchor holds all four keys, in blocks of C_LaunchFieldCount (layout note in
    // Constants). A key whose block the array does not reach keeps what ResetLaunchConfig left —
    // ungranted, which is also what a map without the metadata gets.
    for (Key, 1, Constants::C_LaunchKeyCount) {
        declare Slot = Key - 1;
        declare Base = Slot * Constants::C_LaunchFieldCount;
        if (Meta_LaunchConfig.count < Base + Constants::C_LaunchFieldCount) {
            continue;
        }

        declare Integer[] Fields = [];
        for (Field, 0, Constants::C_LaunchFieldCount - 1) {
            Fields.add(Meta_LaunchConfig[Base + Field]);
        }

        LaunchOn[Slot] = Fields[Constants::C_LaunchFieldOn] == 1;
        LaunchUp[Slot] = ML::Clamp(Fields[Constants::C_LaunchFieldUp], 0, 9);
        LaunchDown[Slot] = ML::Clamp(Fields[Constants::C_LaunchFieldDown], 0, 9);
        LaunchSide[Slot] = ML::Clamp(Fields[Constants::C_LaunchFieldSide], 0, 9);
        LaunchDir[Slot] = ML::Clamp(Fields[Constants::C_LaunchFieldDir], 0, 7);
        LaunchFrame[Slot] = ML::Clamp(Fields[Constants::C_LaunchFieldFrame], 0, Constants::C_LaunchFrameMax);
        LaunchCooldown[Slot] = ML::Clamp(Fields[Constants::C_LaunchFieldCooldown], 0, Constants::C_LaunchCooldownMax);
        LaunchGround[Slot] = Fields[Constants::C_LaunchFieldGround] == 1;
        LaunchAmount[Slot] = ML::Clamp(Fields[Constants::C_LaunchFieldAmount], 0, Constants::C_LaunchAmountMax);
        LaunchGain[Slot] = ML::Clamp(Fields[Constants::C_LaunchFieldGain], 0, Constants::C_LaunchGainMax);
        LaunchRefill[Slot] = Fields[Constants::C_LaunchFieldRefill] == 1;
        LaunchEmpty[Slot] = Fields[Constants::C_LaunchFieldEmpty] == 1;
    }

    InitLaunchCharges(_Player, PreviousOn, PreviousCharges, _Respawned);

    // What the per-tick work is gated on: a section granting no key at all costs nothing per tick
    foreach (Granted in LaunchOn) {
        if (Granted) {
            LaunchAnyOn = True;
        }
    }

    // The strip's static half is settled now and cannot change again until the next section, so
    // publish it here rather than rebuilding it every tick. The charge half follows so a section
    // that revokes every key clears the bars on the same pass.
    LaunchKeys::PublishConfig(_Player);
    LaunchKeys::PublishCharges(_Player);
}

// Refill the charges a player has spent, one key at a time. The charge in progress is a deadline
// rather than an accumulator, so nothing drifts when the tick rate wobbles and the HUD can draw
// the partial bar from the same two numbers.
Void TickLaunchCharges(CSmPlayer _Player) {
    declare Boolean[] LaunchOn for _Player = [];
    declare Integer[] LaunchAmount for _Player = [];
    declare Integer[] LaunchGain for _Player = [];
    declare Integer[] LaunchCharges for _Player = [];
    declare Integer[] LaunchChargeAt for _Player = [];
    if (LaunchOn.count < Constants::C_LaunchKeyCount || LaunchCharges.count < Constants::C_LaunchKeyCount) {
        return;
    }

    for (Key, 1, Constants::C_LaunchKeyCount) {
        declare Index = Key - 1;
        // Ungranted, unlimited (Amount 0), gain 0 or already full: nothing is charging
        declare Charging = LaunchOn[Index] && LaunchAmount[Index] > 0 && LaunchGain[Index] > 0
            && LaunchCharges[Index] < LaunchAmount[Index];
        if (!Charging) {
            LaunchChargeAt[Index] = 0;
            continue;
        }

        if (LaunchChargeAt[Index] <= 0) {
            // Idle with room to fill and no deadline running: a section handed the player a
            // part-spent key. Same full cycle as everywhere else, so entering a section is not a
            // way to skip the hold.
            LaunchChargeAt[Index] = Now + Tools::LaunchCycleMs(LaunchGain[Index]);
        } else if (Now >= LaunchChargeAt[Index]) {
            LaunchCharges[Index] += 1;
            if (LaunchCharges[Index] < LaunchAmount[Index]) {
                // A charge did just complete: the gauge holds before the next one starts
                LaunchChargeAt[Index] = Now + Tools::LaunchCycleMs(LaunchGain[Index]);
            } else {
                LaunchChargeAt[Index] = 0;
            }
        }
    }
}

// Per-tick upkeep for one player's launch keys, and the gate in front of it: a section granting
// no key at all costs nothing here. LoadLaunchConfig has already published the zeroes for those
// players, so there is nothing left to keep up to date.
Void TickLaunchKeys(CSmPlayer _Player) {
    declare Boolean LaunchAnyOn for _Player = False;
    if (!LaunchAnyOn) {
        return;
    }
    TickLaunchCharges(_Player);
    LaunchKeys::PublishCharges(_Player);
}

// What the mode knows about the action instances that actually FIRE a launch, indexed by fire
// slot: 0 is the pad slot (Slot_D) and 1..C_LaunchKeyCount the key slots, so these two are one
// longer than the per-key arrays above. LaunchSlotArmedAt is the Now a launch was handed to the
// slot (0 = nothing waiting on a fire echo); LaunchSlotRearmPass is the first loop pass on which
// it may be armed again. Together they keep one instance to one launch at a time — the rule
// C_LaunchSlotRearmPasses spells out — and they are the whole record of it: what is outstanding,
// and in what order, is read back off the arm stamps rather than tracked a second time. Called
// before any indexed write; the paths that only read or only clear (LaunchSlotReady,
// ClearLaunchSlotState) take a short array as "free" instead, so they stay cheap.
Void EnsureLaunchSlotState(CSmPlayer _Player) {
    declare Integer[] LaunchSlotArmedAt for _Player = [];
    declare Integer[] LaunchSlotRearmPass for _Player = [];
    if (LaunchSlotArmedAt.count > Constants::C_LaunchKeyCount
        && LaunchSlotRearmPass.count > Constants::C_LaunchKeyCount) {
        return;
    }
    LaunchSlotArmedAt = [];
    LaunchSlotRearmPass = [];
    for (Slot, 0, Constants::C_LaunchKeyCount) {
        LaunchSlotArmedAt.add(0);
        LaunchSlotRearmPass.add(0);
    }
}

// Forget what was believed about a fire slot because the instance behind it has just been
// replaced: a fresh one is disarmed with its fire latch already clear, so it can take a launch
// immediately. _SlotIndex < 0 clears every slot.
Void ClearLaunchSlotState(CSmPlayer _Player, Integer _SlotIndex) {
    declare Integer[] LaunchSlotArmedAt for _Player = [];
    declare Integer[] LaunchSlotRearmPass for _Player = [];
    // Empty already means what clearing means: LaunchSlotReady reads a short array as "nothing
    // armed, nothing waiting out a disarm", and EnsureLaunchSlotState sizes them again on the next
    // arm. So every slot at once is dropping the arrays, and one slot of an array that was never
    // sized is already clear.
    if (_SlotIndex < 0) {
        LaunchSlotArmedAt = [];
        LaunchSlotRearmPass = [];
    } else if (LaunchSlotArmedAt.count > _SlotIndex && LaunchSlotRearmPass.count > _SlotIndex) {
        LaunchSlotArmedAt[_SlotIndex] = 0;
        LaunchSlotRearmPass[_SlotIndex] = 0;
    }
}

// Hand a fire slot back: forget the launch it was holding, then keep it shut for a pass so the
// action gets a read of its own on that 0 before anything arms over it. Skipping the hold is what
// silenced the action under rapid fire; C_LaunchSlotRearmPasses has the full story. Both ways a
// launch can end — the fire echo and the lost-echo timeout — resolve the slot through here, which
// is what keeps the two paths from drifting apart.
Void ReleaseLaunchSlot(CSmPlayer _Player, Integer _SlotIndex) {
    EnsureLaunchSlotState(_Player);
    declare Integer[] LaunchSlotArmedAt for _Player = [];
    declare Integer[] LaunchSlotRearmPass for _Player = [];
    LaunchSlotArmedAt[_SlotIndex] = 0;
    LaunchSlotRearmPass[_SlotIndex] = G_LaunchLoopPass + Constants::C_LaunchSlotRearmPasses;
    ActionSetVariant(_Player, LaunchSlot(_SlotIndex), 0);
}

// May this fire slot be handed a launch right now? Both halves of the answer are about the one
// action instance behind the slot, never about the trigger in front of it — pad cooldowns, key
// cooldowns and charges are all still the caller's business. Not a pure test: a slot found timed
// out is recovered here, which is also why every caller asks before it spends anything.
Boolean LaunchSlotReady(CSmPlayer _Player, Integer _SlotIndex) {
    declare Integer[] LaunchSlotArmedAt for _Player = [];
    declare Integer[] LaunchSlotRearmPass for _Player = [];
    // Too short to hold this slot means never written for this player (EnsureLaunchSlotState only
    // ever sizes them in full), so nothing is armed and nothing is waiting out a disarm: free.
    // Answered rather than sized, so the hot reject path this sits on costs the two lookups above
    // and nothing else — the writers below are what actually fill the arrays in.
    if (LaunchSlotArmedAt.count <= _SlotIndex || LaunchSlotRearmPass.count <= _SlotIndex) {
        return True;
    }
    if (LaunchSlotArmedAt[_SlotIndex] > 0) {
        // A launch is still sitting in there unfired; overwriting it would drop it silently.
        if (Now < LaunchSlotArmedAt[_SlotIndex] + Constants::C_LaunchSlotArmTimeoutMs) {
            return False;
        }
        // Its fire echo never came back (C_LaunchSlotArmTimeoutMs). Recover the way a real echo
        // would instead of merely releasing the slot: if the action did fire and only the echo
        // went missing, its latch is still set, and nothing but a zeroed pass of its own clears
        // that — release alone would arm straight back over the 0 and wedge the instance.
        ReleaseLaunchSlot(_Player, _SlotIndex);
        return False;
    }
    // Disarmed, but the action has not necessarily read that 0 back yet — and until it does, its
    // fire latch is still set and the next arm would wedge it (see C_LaunchSlotRearmPasses).
    return G_LaunchLoopPass >= LaunchSlotRearmPass[_SlotIndex];
}

// Bound key = granted ability. Unbinding is what revokes it at the next checkpoint.
//
// A slot is only touched when its granted state actually changes: re-creating an instance is
// not free (it resets the action's own state and costs an echo), and this runs on every section
// change. A map or section with no launch abilities makes no action calls at all.
Void ApplyLaunchBinds(CSmPlayer _Player) {
    declare Boolean[] LaunchOn for _Player = [];
    declare Boolean[] LaunchBound for _Player = [];
    if (LaunchBound.count < Constants::C_LaunchKeyCount) {
        LaunchBound = Tools::LaunchFalseSlots();
    }

    for (Key, 1, Constants::C_LaunchKeyCount) {
        // The key's own model or no binding at all. Its name is the only thing that identifies
        // the key when the press echo arrives (LaunchKeyFromActionId), so a slot loaded with any
        // other model is bound but unplaceable: the HUD offers the ability, the echo comes back
        // naming a model that is not a key, and the press is dropped. Treating the key as
        // ungranted instead at least leaves the HUD and the mode saying the same thing.
        //
        // Asked per key rather than once on G_TrampolineActionId, which gates the pad slot and
        // says nothing about whether this key's model registered.
        declare Ident ModelId = NullId;
        if (G_LaunchActionIds.count >= Key) {
            ModelId = G_LaunchActionIds[Key - 1];
        }
        declare Granted = ModelId != NullId && LaunchOn.count >= Key && LaunchOn[Key - 1];
        if (Granted == LaunchBound[Key - 1]) {
            continue;
        }
        LaunchBound[Key - 1] = Granted;
        if (Granted) {
            ActionLoad(_Player, LaunchSlot(Key), ModelId);
            ActionBind(_Player, LaunchSlot(Key), LaunchInput(Key));
            // Disarmed, so a fresh instance can't fire on whatever the slot held before
            ActionSetVariant(_Player, LaunchSlot(Key), 0);
            // ...and the instance this replaced took its pending launch and its fire latch with
            // it, so nothing about the old one should still gate the new one
            ClearLaunchSlotState(_Player, Key);
        } else {
            ActionBind(_Player, LaunchSlot(Key), CSmMode::EActionInput::None);
        }
    }
}

// A spawn destroys the loaded instances and their bindings (and corrupts what survives), so
// everything the mode believes about this player's slots is stale. The section refresh that
// follows rebinds whatever the new section grants; the pad slot is replaced here because a
// spawn corrupts it whether or not any launch key is involved.
Void LoadLaunchActions(CSmPlayer _Player) {
    declare Boolean[] LaunchBound for _Player = [];
    LaunchBound = Tools::LaunchFalseSlots();
    // Fresh instances: nothing armed in them and no fire latch to wait out
    ClearLaunchSlotState(_Player, -1);
    LoadTrampolineAction(_Player);
}

// Close the window and forget what it was holding. The window is per-map — it restores the
// UsePvPWeapons of the map it was opened on — so it has to be dropped wherever that value is
// decided again, or a launch outstanding at map end restores the old map's setting over the new
// map's (Match_StartMap), and a map left mid-launch carries a phantom in-flight count into the
// next server round (Match_StartServer).
Void ResetLaunchPvPWindow() {
    G_TrampolinePvPWindowEnd = 0;
    G_TrampolinePvPRestore = False;
    G_TrampolinePvPInFlight = 0;
}

// Open the window in which a blast may push players (G_LastUsePvPWeapons keeps the "used PvP
// weapons" run-stamp suppressed — a mode-sanctioned launch is not player PvP). Opened at the
// safety cap; the fire echo collapses it to a short tail once the blast lands.
Void OpenLaunchPvPWindow() {
    if (G_TrampolinePvPWindowEnd <= 0) {
        G_TrampolinePvPRestore = UsePvPWeapons;
    }
    G_TrampolinePvPWindowEnd = Now + Constants::C_TrampolinePvPMaxMs;
    UsePvPWeapons = True;
    G_LastUsePvPWeapons = True;
}

// Arm one launch: open the PvP window, remember which slot is waiting to fire, then hand the
// variant to the action.
Void ArmLaunch(CSmPlayer _Player, Integer _SlotIndex, Integer _Variant) {
    // Backstop, not the gate: callers ask LaunchSlotReady themselves so a launch that cannot go
    // out yet costs nothing (no charge, no cooldown stamp). Repeating the test here is what makes
    // it impossible for a caller that forgets to wedge the instance.
    if (!LaunchSlotReady(_Player, _SlotIndex)) {
        return;
    }

    OpenLaunchPvPWindow();
    G_TrampolinePvPInFlight += 1;

    EnsureLaunchSlotState(_Player);
    declare Integer[] LaunchSlotArmedAt for _Player = [];
    LaunchSlotArmedAt[_SlotIndex] = Now;	// busy until the fire echo comes back (or the timeout)

    ActionSetVariant(_Player, LaunchSlot(_SlotIndex), _Variant);
}

// One armed launch has resolved: retire it from the in-flight count. Once nothing else is still
// waiting to fire, collapse the global PvP window to a short post-fire tail (bullets explode
// immediately, so the knockback is landing now) instead of holding it open for the safety cap.
Void RetireLaunchInFlight() {
    if (G_TrampolinePvPInFlight > 0) {
        G_TrampolinePvPInFlight -= 1;
    }
    if (G_TrampolinePvPInFlight <= 0 && G_TrampolinePvPWindowEnd > 0) {
        declare ShortEnd = Now + Constants::C_TrampolinePvPPostFireMs;
        if (ShortEnd < G_TrampolinePvPWindowEnd) {
            G_TrampolinePvPWindowEnd = ShortEnd;
        }
    }
}

// A granted key was pressed: check the section's gates, then arm that key's slot exactly the
// way a pad touch arms Slot_D.
Void TryLaunchOnKey(CSmPlayer _Player, Integer _Key) {
    if (_Player.SpawnStatus != CSmPlayer::ESpawnStatus::Spawned) {
        return;
    }

    declare Boolean[] LaunchOn for _Player = [];
    if (LaunchOn.count < _Key || !LaunchOn[_Key - 1]) {
        return;
    }
    declare Index = _Key - 1;

    // The three rejecting gates come before the rest of the slots are read, because this is the
    // hot path: a held key echoes every tick (see Actions/Trampoline.Script.txt) and almost
    // every echo dies right here — 49 of 50 at the default 500 ms cooldown, 149 of 150 while a
    // key is waiting on a charge. Reading the other slots below them keeps a discarded echo to
    // five persistent lookups instead of twelve. Free to do: all three gates only return, and
    // everything with a side effect (the charge spend, ArmLaunch, the ReadyAt stamp) stays
    // after them, so nothing observes the order.
    //
    // Cooldown leads the three because it is the gate that does the rejecting. Ground-gating is
    // the rare configuration, so on a key that does not use it the slot below never decides
    // anything, and putting the cooldown first means a discarded echo does not read it at all.
    declare Integer[] LaunchReadyAt for _Player = [];
    if (Now < LaunchReadyAt[Index]) {
        return;
    }
    declare Boolean[] LaunchGround for _Player = [];
    if (LaunchGround[Index] && !_Player.IsTouchingGround) {
        return;
    }
    // Which instance fires this launch (see C_LaunchFireOnPadSlot): the shared pad slot by
    // default, the key's own slot otherwise. Needed here rather than at the ArmLaunch call
    // because the instance being free is a gate like the two above — and one that must reject
    // in front of every side effect, so a press the slot cannot take yet neither eats a charge
    // nor stamps a cooldown. The key echoes again next tick, so the launch simply goes out on
    // the first tick the slot can take it.
    declare FireSlot = _Key;
    if (Constants::C_LaunchFireOnPadSlot) {
        FireSlot = 0;
    }
    if (!LaunchSlotReady(_Player, FireSlot)) {
        return;
    }

    declare Integer[] LaunchUp for _Player = [];
    declare Integer[] LaunchDown for _Player = [];
    declare Integer[] LaunchSide for _Player = [];
    declare Integer[] LaunchDir for _Player = [];
    declare Integer[] LaunchFrame for _Player = [];
    declare Integer[] LaunchCooldown for _Player = [];
    declare Integer[] LaunchAmount for _Player = [];
    declare Integer[] LaunchGain for _Player = [];
    declare Integer[] LaunchCharges for _Player = [];
    declare Integer[] LaunchChargeAt for _Player = [];

    // Policy: optionally suppress the down blast when an up blast fires too
    declare DownStrength = Tools::LaunchDownStrength(LaunchUp[Index], LaunchDown[Index]);
    if (LaunchUp[Index] <= 0 && DownStrength <= 0 && LaunchSide[Index] <= 0) {
        return;
    }

    // Charges, once every gate a launch could still fail on has passed — a key configured to do
    // nothing must not eat one. Amount 0 is unlimited, so it only stamps the cooldown below.
    if (LaunchAmount[Index] > 0) {
        if (LaunchCharges[Index] <= 0) {
            return;
        }
        LaunchCharges[Index] -= 1;
        // Open a cycle on the slot this shot just emptied when the key was full, and leave a
        // charge that is already part-way there alone: firing costs a charge, never the progress
        // towards one. The full cycle, hold included — the slot a shot empties comes back exactly
        // as fast as one the gauge emptied itself, so the first shot off a full gauge is not
        // quietly cheaper than the ones after it.
        if (LaunchChargeAt[Index] <= 0 && LaunchGain[Index] > 0) {
            LaunchChargeAt[Index] = Now + Tools::LaunchCycleMs(LaunchGain[Index]);
        }
    }

    // Encoding v4 and the frame's own direction calibration, both in LaunchVariant
    declare ArmedVariant = LaunchVariant(LaunchFrame[Index],
        LaunchUp[Index], LaunchDown[Index], LaunchDir[Index], LaunchSide[Index]);
    ArmLaunch(_Player, FireSlot, ArmedVariant);
    LaunchReadyAt[Index] = Now + LaunchCooldown[Index];
}

Void UpdateSettings() {
    if (S_TimeLimit != Settings::GetTimeLimit()) {
        Settings::SetTimeLimit(S_TimeLimit);

        if (Hud != Null && Hud.ScoresTable != Null) {
            Hud.ScoresTable.SetFooterText(TL::Compose("%1 "^Tools::TimeLimitToText(S_TimeLimit), _("Time limit : ")));
        }

        if (S_TimeLimit > 0) {
            EndTime = StartTime + (S_TimeLimit * 60 * 1000);
        } else {
            EndTime = -1;
        }
    }

    if (S_FastRespawn != Settings::GetFastRespawn()) {
        Settings::SetFastRespawn(S_FastRespawn);
    }

    if (S_AllowJump != Settings::GetAllowJump()) {
        Settings::SetAllowJump(S_AllowJump);
    }

    if (!Settings::GetForceDisableRecords() && (S_EnableRecords != Settings::GetEnableRecords() || MapAttr::IsLeaderboardEnabled() != Settings::GetEnableRecords())) {
        Settings::SetEnableRecords(S_EnableRecords && MapAttr::IsLeaderboardEnabled());
    }

    if (S_NextMapVotingRatio != Settings::GetNextMapVotingRatio()) {
        Settings::SetNextMapVotingRatio(S_NextMapVotingRatio);
    }

    if (S_FullAmmoSpawn != Settings::GetFullAmmoSpawn()) {
        Settings::SetFullAmmoSpawn(S_FullAmmoSpawn);
    }

    if (S_LegacyRespawn != Settings::GetLegacyRespawn()) {
        Settings::SetLegacyRespawn(S_LegacyRespawn);
    }

    if (S_BestNewRecordMsg != Settings::GetBestNewRecordMsg()) {
        Settings::SetBestNewRecordMsg(S_BestNewRecordMsg);
    }

    if (S_WorseNewRecordMsg != Settings::GetWorseNewRecordMsg()) {
        Settings::SetWorseNewRecordMsg(S_WorseNewRecordMsg);
    }

    declare Boolean ForceInvisibleSkin = S_ForceInvisibleSkin && G_ForceInvisibleSkin;
    if (ForceInvisibleSkin != Settings::GetForceInvisibleSkin()) {
        Settings::SetForceInvisibleSkin(ForceInvisibleSkin);

        if (Settings::GetForceInvisibleSkin()) {
            Obstacle::UpdatePlayersSkin(G_InvisibleSkinId);
        } else {
            Obstacle::UpdatePlayersSkin(NullId);
        }
    }

    if (S_EnablePlayerHighlight != Settings::GetEnablePlayerHighlight()) {
        Settings::SetEnablePlayerHighlight(S_EnablePlayerHighlight);

        if (Settings::GetForceInvisibleSkin()) {
            Obstacle::UpdatePlayersSkin(G_InvisibleSkinId);
        } else {
            Obstacle::UpdatePlayersSkin(NullId);
        }
    }

    if (S_ForceRNGSeed != Settings::GetForceRNGSeed()) {
        Settings::SetForceRNGSeed(S_ForceRNGSeed);
    }
    
    if (S_PvPWeapons != Settings::GetForcePvPWeapons()) {
        Settings::SetForcePvPWeapons(S_PvPWeapons);
        UsePvPWeapons = S_PvPWeapons || Obstacle::IsPvPWeaponsMap();
    }

    if (S_PvPCollisions != Settings::GetForcePvPCollisions()) {
        Settings::SetForcePvPCollisions(S_PvPCollisions);
        UsePvPCollisions = S_PvPCollisions || Obstacle::IsPvPCollisionsMap();
    }

    if (S_CheckpointMarkersCount != Settings::GetCheckpointMarkersCount()) {
        Settings::SetCheckpointMarkersCount(S_CheckpointMarkersCount);
        Layers::Update("Markers", Markers::GetLayer(S_CheckpointMarkersCount));
    }

    // --- Event settings ---

    if (!ObsEvents::HasForcedClear()) {
        if (ObsEvents::GetEventHandle() == "" && S_EventHandle != ""
            && ObsEvents::GetEventEdition() <= 0 && S_EventEdition > 0)
        {
            ObsEvents::SetEventHandle(S_EventHandle);
            ObsEvents::SetEventEdition(S_EventEdition);
        }
    }

    if (ObsEvents::GetEventHandle() != Settings::GetEventHandle()) {
        Settings::SetEventHandle(ObsEvents::GetEventHandle());
    }

    if (ObsEvents::GetEventEdition() != Settings::GetEventEdition()) {
        Settings::SetEventEdition(ObsEvents::GetEventEdition());
    }

    if (S_SwitchEventVotingRatio != Settings::GetSwitchEventVotingRatio()) {
        Settings::SetSwitchEventVotingRatio(S_SwitchEventVotingRatio);
    }

    if (S_LockEventsUI != Settings::GetLockEventsUI()) {
        Settings::SetLockEventsUI(S_LockEventsUI);
    }

    if (S_TimeBeforeSwitchEvent != Settings::GetTimeBeforeSwitchEvent()) {
        Settings::SetTimeBeforeSwitchEvent(S_TimeBeforeSwitchEvent);
    }
}
